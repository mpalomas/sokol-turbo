function(sokol_apply_release_size_flags target)
    if(CMAKE_BUILD_TYPE MATCHES "^[Rr]elease$")
        if(MSVC)
            target_compile_options(${target} PRIVATE /Gy)
            target_link_options(${target} PRIVATE /OPT:REF /OPT:ICF)
        elseif(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang|AppleClang")
            target_compile_options(${target} PRIVATE -ffunction-sections -fdata-sections)
            if(APPLE)
                target_link_options(${target} PRIVATE -dead_strip)
            else()
                target_link_options(${target} PRIVATE -Wl,--gc-sections)
            endif()
        endif()
    endif()
endfunction()

function(sokol_configure_target target library_type)
    target_include_directories(${target} PUBLIC
        "$<BUILD_INTERFACE:${SOKOL_ROOT}>"
        "$<BUILD_INTERFACE:${SOKOL_ROOT}/util>"
        "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>"
        "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/util>")
    target_compile_definitions(${target} PUBLIC ${sokol_public_defines})
    if(library_type STREQUAL "SHARED")
        target_compile_definitions(${target} PUBLIC SOKOL_DLL)
    endif()
    sokol_apply_release_size_flags(${target})
endfunction()

function(sokol_add_component target library_type source)
    add_library(${target} ${library_type} "${source}")
    sokol_configure_target(${target} ${library_type})
    target_compile_definitions(${target} PRIVATE IMPL)
    target_link_libraries(${target} PUBLIC ${ARGN} ${sokol_system_libs})
    if(target STREQUAL "sokol_app_shared")
        target_compile_definitions(${target} PUBLIC SOKOL_NO_ENTRY)
    endif()
    if(APPLE AND source MATCHES "sokol_app")
        set_source_files_properties("${source}" PROPERTIES LANGUAGE OBJC)
    endif()
endfunction()

macro(sokol_component option target source)
    if(${option})
        set(static_deps)
        set(shared_deps)
        foreach(dep ${ARGN})
            list(APPEND static_deps ${dep}_static)
            list(APPEND shared_deps ${dep}_shared)
        endforeach()
        if(SOKOL_BUILD_STATIC)
            sokol_add_component(${target}_static STATIC "${source}" ${static_deps})
            list(APPEND sokol_static_components ${target}_static)
        endif()
        if(SOKOL_BUILD_SHARED)
            sokol_add_component(${target}_shared SHARED "${source}" ${shared_deps})
            list(APPEND sokol_shared_components ${target}_shared)
        endif()
    endif()
endmacro()

function(sokol_sample_target_name out name suffix)

endfunction()

function(sokol_resolve_shdc)
    if(SOKOL_SHDC)
        set(SOKOL_SHDC_EXE "${SOKOL_SHDC}" PARENT_SCOPE)
        return()
    endif()
    FetchContent_Declare(sokol_tools_bin
        GIT_REPOSITORY https://github.com/floooh/sokol-tools-bin.git
        GIT_TAG master
        UPDATE_DISCONNECTED TRUE
        GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(sokol_tools_bin)
    if(CMAKE_HOST_SYSTEM_NAME STREQUAL "Linux")
        set(host_dir "linux")
    elseif(CMAKE_HOST_SYSTEM_NAME STREQUAL "Darwin")
        set(host_dir "osx")
    elseif(CMAKE_HOST_SYSTEM_NAME STREQUAL "Windows")
        set(host_dir "win32")
        set(host_suffix ".exe")
    else()
        message(FATAL_ERROR "Unsupported host for automatic sokol-shdc: ${CMAKE_HOST_SYSTEM_NAME}")
    endif()
    set(SOKOL_SHDC_EXE "${sokol_tools_bin_SOURCE_DIR}/bin/${host_dir}/sokol-shdc${host_suffix}" PARENT_SCOPE)
endfunction()

function(sokol_shader_lang out)
    if(SOKOL_BACKEND STREQUAL "SOKOL_GLCORE")
        set(lang glsl410)
    elseif(SOKOL_BACKEND STREQUAL "SOKOL_GLES3")
        set(lang glsl300es)
    elseif(SOKOL_BACKEND STREQUAL "SOKOL_D3D11")
        set(lang hlsl4)
    elseif(SOKOL_BACKEND STREQUAL "SOKOL_METAL")
        if(APPLE)
            set(lang metal_macos)
        else()
            set(lang metal_macos)
        endif()
    elseif(SOKOL_BACKEND STREQUAL "SOKOL_WGPU")
        set(lang wgsl)
    elseif(SOKOL_BACKEND STREQUAL "SOKOL_VULKAN")
        set(lang spirv_vk)
    else()
        set(lang glsl410)
    endif()
    set(${out} "${lang}" PARENT_SCOPE)
endfunction()

function(sokol_generate_shader out_files sample)
    sokol_resolve_shdc()
    sokol_shader_lang(shader_lang)
    set(src "${SOKOL_SAMPLES_DIR}/sapp/${sample}-sapp.glsl")
    if(sample STREQUAL "shdfeatures")
        set(result)
        foreach(item "none;NONE" "slm;SKINNING,LIGHTING,MATERIAL" "sl;SKINNING,LIGHTING" "s;SKINNING" "sm;SKINNING,MATERIAL" "lm;LIGHTING,MATERIAL" "m;MATERIAL" "l;LIGHTING")
            string(REPLACE ";" "|" pair "${item}")
            string(REPLACE "|" ";" pair "${pair}")
            list(GET pair 0 mod)
            list(GET pair 1 defs)
            set(out "${CMAKE_CURRENT_BINARY_DIR}/samples/sapp/${sample}-sapp.glsl.${mod}.h")
            add_custom_command(
                OUTPUT "${out}"
                COMMAND "${CMAKE_COMMAND}" -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/samples/sapp"
                COMMAND "${SOKOL_SHDC_EXE}" --input "${src}" --output "${out}" --slang "${shader_lang}" --module "${mod}" --reflection --defines "${defs}"
                DEPENDS "${src}" "${SOKOL_SHDC_EXE}"
                VERBATIM)
            list(APPEND result "${out}")
        endforeach()
    else()
        set(out "${CMAKE_CURRENT_BINARY_DIR}/samples/sapp/${sample}-sapp.glsl.h")
        add_custom_command(
            OUTPUT "${out}"
            COMMAND "${CMAKE_COMMAND}" -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/samples/sapp"
            COMMAND "${SOKOL_SHDC_EXE}" --input "${src}" --output "${out}" --slang "${shader_lang}"
            DEPENDS "${src}" "${SOKOL_SHDC_EXE}"
            VERBATIM)
        set(result "${out}")
    endif()
    set(${out_files} "${result}" PARENT_SCOPE)
endfunction()

function(sokol_add_sample name)
    set(options SHD COMPUTE)
    set(one_value_args EXT UI)
    set(multi_value_args DEPS)
    cmake_parse_arguments(SAMPLE "${options}" "${one_value_args}" "${multi_value_args}" ${ARGN})
    string(TOUPPER "${name}" opt_name)
    string(REPLACE "-" "_" opt_name "${opt_name}")
    set(sample_option "SOKOL_SAMPLE_${opt_name}")
    option(${sample_option} "Build ${name}-sapp sample" ON)
    if(NOT ${sample_option})
        return()
    endif()
    if(SAMPLE_COMPUTE AND (SOKOL_BACKEND STREQUAL "SOKOL_GLES3" OR (APPLE AND SOKOL_BACKEND STREQUAL "SOKOL_GLCORE")))
        return()
    endif()
    foreach(dep IN LISTS SAMPLE_DEPS)
        if(dep STREQUAL "imgui" OR dep STREQUAL "imgui-docking")
            if(NOT SOKOL_SAMPLES_WITH_IMGUI)
                return()
            endif()
        elseif(dep STREQUAL "nuklear-static")
            if(NOT SOKOL_SAMPLES_WITH_NUKLEAR)
                return()
            endif()
        elseif(dep STREQUAL "basisu")
            if(NOT SOKOL_SAMPLES_WITH_BASISU)
                return()
            endif()
        elseif(dep STREQUAL "spine")
            if(NOT SOKOL_SAMPLES_WITH_SPINE)
                return()
            endif()
        elseif(dep STREQUAL "ozzanimation")
            if(NOT SOKOL_SAMPLES_WITH_OZZ)
                return()
            endif()
        elseif(dep STREQUAL "ozzutil")
            if(NOT SOKOL_SAMPLES_WITH_OZZ)
                return()
            endif()
        elseif(dep STREQUAL "libmodplug")
            if(NOT SOKOL_SAMPLES_WITH_MODPLUG)
                return()
            endif()
        elseif(dep STREQUAL "microui")
            if(NOT SOKOL_SAMPLES_WITH_MICROUI)
                return()
            endif()
        elseif(dep STREQUAL "stb")
            return()
        endif()
    endforeach()
    set(ext "${SAMPLE_EXT}")
    if(ext STREQUAL "")
        set(ext c)
    endif()
    set(src "${SOKOL_SAMPLES_DIR}/sapp/${name}-sapp.${ext}")
    if(NOT EXISTS "${src}")
        message(WARNING "Skipping ${name}-sapp, missing ${src}")
        return()
    endif()
    set(generated)
    if(SAMPLE_SHD)
        sokol_generate_shader(generated "${name}")
    endif()
    sokol_sample_target_name(target "${name}" "")
    if(APPLE)
        add_executable(${target} MACOSX_BUNDLE "${src}" ${generated})
    elseif(WIN32)
        add_executable(${target} WIN32 "${src}" ${generated})
    else()
        add_executable(${target} "${src}" ${generated})
    endif()
    target_include_directories(${target} PRIVATE
        "${SOKOL_SAMPLES_DIR}/sapp"
        "${SOKOL_SAMPLES_DIR}/libs"
        "${SOKOL_SAMPLES_DIR}/libs/util"
        "${SOKOL_SAMPLES_DIR}/libs/vecmath"
        "${SOKOL_SAMPLES_DIR}/libs/cgltf"
        "${SOKOL_SAMPLES_DIR}/libs/pl_mpeg"
        "${SOKOL_SAMPLES_DIR}/libs/qoi"
        "${CMAKE_CURRENT_BINARY_DIR}/samples/sapp")
    target_link_libraries(${target} PRIVATE sokol)
    sokol_apply_release_size_flags(${target})
    foreach(dep IN LISTS SAMPLE_DEPS)
        if(TARGET sample_${dep})
            target_link_libraries(${target} PRIVATE sample_${dep})
        endif()
    endforeach()
    add_custom_command(TARGET ${target} POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E copy_directory_if_different
            "${SOKOL_SAMPLES_DIR}/sapp/data"
            "$<TARGET_FILE_DIR:${target}>/data"
        VERBATIM)
    if(SOKOL_STRIP_RELEASE_BINARIES AND CMAKE_BUILD_TYPE MATCHES "^[Rr]elease$" AND CMAKE_STRIP)
        add_custom_command(TARGET ${target} POST_BUILD COMMAND "${CMAKE_STRIP}" "$<TARGET_FILE:${target}>")
    endif()
    if(SAMPLE_UI AND SOKOL_SAMPLES_WITH_IMGUI AND TARGET sample_dbgui)
        sokol_sample_target_name(ui_target "${name}" "_ui")
        add_executable(${ui_target} "${src}" ${generated})
        target_compile_definitions(${ui_target} PRIVATE USE_DBG_UI=1)
        target_include_directories(${ui_target} PRIVATE
            "${SOKOL_SAMPLES_DIR}/sapp"
            "${SOKOL_SAMPLES_DIR}/libs"
            "${CMAKE_CURRENT_BINARY_DIR}/samples/sapp")
        target_link_libraries(${ui_target} PRIVATE sokol sample_dbgui)
        sokol_apply_release_size_flags(${ui_target})
    endif()
endfunction()
